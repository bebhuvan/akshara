[\s\S]*?

|

[\s\S]*?

` $html 6 -}} {{- if not $out -}} {{- $txt := . | replaceRE `` "" | replaceRE `` "" | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- $caps := len (findRE `[A-Z]` $txt) -}}{{- $lows := len (findRE `[a-z]` $txt) -}} {{- if and (gt (len $txt) 60) (gt $lows (mul $caps 2)) -}} {{- with findRE `^[\s\S]{40,400}?[.!?](?:["’”])?(?:\s|$)` $txt 1 -}}{{- $out = strings.TrimSpace (index . 0) -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
