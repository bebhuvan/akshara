{{- /* The catalogue grouped one of three ways: "shelf", "year" (by decade) or "author" (by initial). Takes dict "books" "order" "cap" (entries per shelf before "and n more"; 0 for none). */ -}} {{- $books := .books -}}{{- $order := .order -}}{{- $cap := .cap | default 0 -}} {{- if eq $order "shelf" -}} {{- range $i, $s := hugo.Data.shelves.shelf -}} {{- $bs := sort (where $books "shelf_no" $i) "year" -}} {{- if $bs -}}

{{ $s.name }}

    {{ range $k, $b := $bs }}{{ partial "book/entry.html" (dict "book" $b "over" (and (gt $cap 0) (ge $k $cap))) }}{{ end }}
{{- if and (gt $cap 0) (gt (len $bs) $cap) }}

and {{ partial "words.html" (sub (len $bs) $cap) }} more on this shelf

{{ end -}}
{{- end -}} {{- end -}} {{- else if eq $order "year" -}} {{- $decades := slice -}} {{- range $books -}}{{- $decades = $decades | append (mul (div (int .year) 10) 10) -}}{{- end -}} {{- range sort (uniq $decades) -}}{{- $d := . -}}

{{ $d }}s

    {{ range sort $books "year" }}{{ if eq (mul (div (int .year) 10) 10) $d }}{{ partial "book/entry.html" (dict "book" .) }}{{ end }}{{ end }}
{{- end -}} {{- else -}} {{- $letters := slice -}} {{- range $books -}}{{- $letters = $letters | append (upper (substr .sort_name 0 1)) -}}{{- end -}} {{- range sort (uniq $letters) -}}{{- $L := . -}}

{{ $L }}

    {{ range sort $books "sort_name" }}{{ if eq (upper (substr .sort_name 0 1)) $L }}{{ partial "book/entry.html" (dict "book" . "inverted" true) }}{{ end }}{{ end }}
{{- end -}} {{- end -}}
