{{- /* One book's record, as every page shows it. Takes a book page (content/books//_index.md) and returns a dict. All hand-set exceptions come from data/bookdetails.toml; everything else from front matter. */ -}} {{- $p := . -}} {{- $id := path.Base (strings.TrimSuffix "/" $p.RelPermalink) -}} {{- $d := index hugo.Data.bookdetails $id | default dict -}} {{- $title := replaceRE `\s+:` ":" $p.Title -}} {{- /* typographer's apostrophes: the pipeline's titles carry straight ones */ -}} {{- $title = replace $title "'" "’" -}} {{- with $p.Params.subtitle -}} {{- with findRESubmatch `^Volume ([IVX]+)\b` . 1 -}} {{- if not (strings.Contains $title "Vol.") -}}{{- $title = printf "%s, Vol. %s" $title (index (index . 0) 1) -}}{{- end -}} {{- end -}} {{- end -}} {{- $cover := $d.cover_title | default "" -}} {{- if not $cover -}} {{- $cover = $title -}} {{- if gt (len $cover) 46 -}}{{- $cover = replaceRE `\s*[:—(].*$` "" $cover -}}{{- end -}} {{- if gt (len $cover) 56 -}}{{- $cover = replaceRE `,[^,]*$` "" $cover -}}{{- end -}} {{- end -}} {{- $author := $d.author | default (replaceRE `\.([A-Z])` ". $1" (replaceRE `^Sir\s+` "" (replaceRE `,\s*C\.I\.E\.` "" ($p.Params.author | default "")))) -}} {{- $sort := $d.sort_name | default "" -}} {{- if not $sort -}} {{- $words := split (strings.TrimSpace (replaceRE ` and .*$` "" (replaceRE `,.*$` "" $author))) " " -}} {{- $last := index $words (sub (len $words) 1) -}} {{- $rest := delimit (first (sub (len $words) 1) $words) " " -}} {{- $sort = cond (eq $rest "") $last (printf "%s, %s" $last $rest) -}} {{- end -}} {{- $shelf := dict -}} {{- $shelfNo := 7 -}} {{- range $i, $s := hugo.Data.shelves.shelf -}}{{- if eq $s.key $p.Params.archetype -}}{{- $shelf = $s -}}{{- $shelfNo = $i -}}{{- end -}}{{- end -}} {{- if not $shelf -}}{{- errorf "Book %s has archetype %q, which is not one of the eight shelves in data/shelves.toml" $id $p.Params.archetype -}}{{- end -}} {{- if not (and $shelf.code $shelf.paper $shelf.ink $shelf.paper_dark $shelf.ink_dark) -}}{{- errorf "Shelf %q in data/shelves.toml needs code, paper, ink, paper_dark and ink_dark" $shelf.name -}}{{- end -}} {{- $no := index hugo.Data.series $id -}} {{- if not $no -}}{{- errorf "Book %s has no series number. Add the next number to data/series.toml" $id -}}{{- end -}} {{- /* The cover (data/covers.toml): its kind, its own picture, and the two lines of its label. */ -}} {{- $cv := index hugo.Data.covers $id -}} {{- if not $cv -}}{{- errorf "Book %s has no cover in data/covers.toml" $id -}}{{- end -}} {{- $kinds := slice "plate" "map" "device" "portrait" "photo" "type" -}} {{- if not (in $kinds $cv.kind) -}}{{- errorf "Book %s: cover kind %q is not one of %v (data/covers.toml)" $id $cv.kind $kinds -}}{{- end -}} {{- if eq $cv.kind "type" -}} {{- with $cv.plate -}}{{- errorf "Book %s: a type-only cover takes no plate (data/covers.toml)" $id -}}{{- end -}} {{- else -}} {{- if not $cv.plate -}}{{- errorf "Book %s: a %s cover needs a plate (data/covers.toml)" $id $cv.kind -}}{{- end -}} {{- if not (fileExists (printf "static/plates/%s.webp" $cv.plate)) -}}{{- errorf "Book %s: plate %q is not in static/plates/" $id $cv.plate -}}{{- end -}} {{- end -}} {{- $cover = replace $cover "'" "’" -}} {{- $lblTitle := $cv.title | default $cover -}} {{- $lblLine := $cv.line | default "" -}} {{- if not $cv.title -}} {{- with findRESubmatch `^(.*), Vol\. ([IVX]+)$` $lblTitle 1 -}}{{- $lblTitle = index (index . 0) 1 -}}{{- $lblLine = printf "Volume %s" (index (index . 0) 2) -}} {{- else -}}{{- with findRESubmatch `^(.*), (\d{4}–\d{4})$` $lblTitle 1 -}}{{- $lblTitle = index (index . 0) 1 -}}{{- $lblLine = index (index . 0) 2 -}} {{- else -}}{{- with findRESubmatch `^(.*): (.{3,22})$` $lblTitle 1 -}}{{- $lblTitle = index (index . 0) 1 -}}{{- $lblLine = index (index . 0) 2 -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}} {{- $coverData := dict "kind" $cv.kind "plate" ($cv.plate | default "") "inset" ($cv.inset | default "") "title" $lblTitle "line" $lblLine -}} {{- $year := printf "%v" $p.Params.year -}} {{- $pages := int ($p.Params.pages | default 300) -}} {{- $tier := $p.Params.edition_tier | default "v1_archive" -}} {{- $tierName := index (dict "machine_checked" "Machine-checked edition" "community" "Community edition, open for proofreading" "v1_archive" "Stage One edition") $tier -}} {{- return dict "page" $p "id" $id "url" $p.RelPermalink "title" $title "cover_title" $cover "subtitle" ($p.Params.subtitle | default "") "author" $author "sort_name" $sort "year" $year "pages" $pages "no" $no "shelf" $shelf "shelf_no" $shelfNo "code" $shelf.code "cover" $coverData "tier" $tier "tier_name" $tierName -}}
